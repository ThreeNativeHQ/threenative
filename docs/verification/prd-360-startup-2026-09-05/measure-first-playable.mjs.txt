#!/usr/bin/env node
// Launch-to-observed-player-movement upper bound for one Bayview arm.
//
// The clock is the coordinator's own monotonic clock. t0 is stamped after the device is known to be
// running the expected APK and the app is force-stopped, and before the playtest runner is spawned;
// the runner force-stops and launches the app itself inside driver.prepare(). t1 is stamped the
// moment the runner exits. The interval therefore contains adb, the mailbox transport, runner
// process startup, the app launch, its asset load, its startup gate, and the step that presses KeyW
// and measures the player. It is an upper bound on launch-to-first-playable, never exact
// process-relative timing, and never smaller than the game's own latency.
//
// Nothing here decides whether a run passed: it records what happened and hands it to
// evaluate-first-playable.mjs.
import { execFileSync, spawn } from 'node:child_process';
import { createHash } from 'node:crypto';
import { existsSync, mkdirSync, readFileSync, writeFileSync } from 'node:fs';
import { dirname, join, resolve } from 'node:path';
import { fileURLToPath } from 'node:url';
import { assertDeviceReady, MINIMUM_BATTERY_PERCENT } from '../../../../packages/runtime-native/scripts/device-preflight.mjs';

const here = dirname(fileURLToPath(import.meta.url));
const repoRoot = resolve(here, '..', '..', '..', '..');
const recovered = join(repoRoot, 'artifacts', 'batch-2026-09-05', 'recovered-bayview-apk-input');
const runnerCli = join(repoRoot, 'packages', 'playtest', 'dist', 'runner', 'cli.js');
const scenario = join(here, 'bayview-first-playable.playtest.json');

// Identities recovered and verified in this lane; see candidate-verification.txt.
const PACKAGE_NAME = 'com.threenative.bayview';
const ACTIVITY = 'com.threenative.runtime.MystralActivity';
const ARMS = {
  candidate: {
    apk: join(here, '..', 'bayview-candidate.apk'),
    sha256: '403bd10c297f91f40202b322d57bc9c3d4291b1f2c9c9a958c1c8b3c5f9053af',
  },
  frozen: {
    apk: undefined, // resolved from the recovery manifest's sourceApk
    sha256: undefined,
  },
};

const usage = () => {
  process.stderr.write(
    'usage: measure-first-playable.mjs --arm <frozen|candidate> --device <serial>\n'
    + '                                 --out <run-dir> [--max-wall-ms <ms>] [--timeout <ms>]\n'
    + '                                 [--skip-install] [--dry-run]\n'
    + '  --out is required: each attempt writes its own run directory (e.g. runs/candidate/a),\n'
    + '  and an existing directory is never overwritten.\n'
    + '  --dry-run never contacts a device.\n',
  );
  process.exit(2);
};

const argv = process.argv.slice(2);
const flag = (name) => {
  const index = argv.indexOf(name);
  return index < 0 ? undefined : argv[index + 1];
};
const has = (name) => argv.includes(name);

const arm = flag('--arm');
const serial = flag('--device');
if (arm !== 'frozen' && arm !== 'candidate') usage();
if (serial === undefined && !has('--dry-run')) usage();

const maxWallMs = Number(flag('--max-wall-ms') ?? 300_000);
const operationTimeoutMs = Number(flag('--timeout') ?? 60_000);
if (!Number.isFinite(maxWallMs) || maxWallMs <= 0) throw new Error('--max-wall-ms must be a positive number');
if (!Number.isFinite(operationTimeoutMs) || operationTimeoutMs <= 0) throw new Error('--timeout must be a positive number');

const outDirFlag = flag('--out');
if (outDirFlag === undefined) usage();
const outDir = resolve(outDirFlag);
const dryRun = has('--dry-run');

// The frozen arm's APK is whatever the recovery manifest recorded, hashed here rather than pinned
// to a literal, so this never claims an identity it did not read.
if (arm === 'frozen') {
  const record = JSON.parse(readFileSync(join(recovered, 'recovery-manifest.json'), 'utf8'));
  ARMS.frozen.apk = record.sourceApk;
  ARMS.frozen.sha256 = createHash('sha256').update(readFileSync(record.sourceApk)).digest('hex');
}

const apk = resolve(ARMS[arm].apk);
const expectedSha256 = ARMS[arm].sha256
  ?? createHash('sha256').update(readFileSync(apk)).digest('hex');

const adb = (() => {
  const candidates = [
    process.env.THREENATIVE_ADB,
    process.env.ANDROID_SDK_ROOT === undefined ? undefined : join(process.env.ANDROID_SDK_ROOT, 'platform-tools', 'adb'),
    process.env.ANDROID_HOME === undefined ? undefined : join(process.env.ANDROID_HOME, 'platform-tools', 'adb'),
    join(process.env.HOME ?? '', 'Android', 'Sdk', 'platform-tools', 'adb'),
  ];
  const found = candidates.find((candidate) => candidate !== undefined && existsSync(candidate));
  if (found === undefined && !dryRun) {
    throw new Error('adb was not found; set THREENATIVE_ADB, ANDROID_SDK_ROOT or ANDROID_HOME');
  }
  return found ?? '<adb>';
})();

const adbArgs = (...rest) => ['-s', serial ?? '<serial>', ...rest];
const planned = [];
const adbRun = (...rest) => {
  const args = adbArgs(...rest);
  planned.push([adb, ...args].join(' '));
  if (dryRun) return '';
  return execFileSync(adb, args, { encoding: 'utf8', maxBuffer: 64 * 1024 * 1024 }).trim();
};

// The local file must be the file this arm claims before anything is installed from it.
const localSha256 = createHash('sha256').update(readFileSync(apk)).digest('hex');
if (localSha256 !== expectedSha256) {
  throw new Error(`${arm} APK at ${apk} hashes ${localSha256}, expected ${expectedSha256}`);
}

const runnerArgs = [
  runnerCli,
  scenario,
  '--target', 'android',
  '--device', serial ?? '<serial>',
  '--package', PACKAGE_NAME,
  '--activity', ACTIVITY,
  '--artifacts', outDir,
  '--timeout', String(operationTimeoutMs),
];

// Fail closed before any performance launch: the device must be qualified (physical, online,
// battery >= 50%, discharging, thermal NONE, screen on) — the same gate the native lanes use —
// and the readings are recorded in bound.json. The gate runs twice: once before the install
// (whose heating must not leave a stale pass) and again immediately before the timed launch,
// so the recorded preflight is the one the timed interval actually started under.
const adbShell = (args) => execFileSync(adb, ['-s', serial, ...args], { encoding: 'utf8', maxBuffer: 8 * 1024 * 1024 }).trim();
const qualifyDevice = async (phase) => {
  const reading = await assertDeviceReady(
    serial,
    {
      allowOverride: false,
      maxThermalStatus: 'NONE',
      minBatteryPercent: MINIMUM_BATTERY_PERCENT,
      requireDischarging: true,
    },
    { adb: (args) => adbShell(args) },
  );
  return { phase, recordedAt: new Date().toISOString(), ...reading };
};

planned.push(['node', ...runnerArgs].join(' '));
if (dryRun) {
  // Dry-run never contacts a device: adbRun only records plans and exits here, before any
  // install, identity read, or qualification. The qualification the command would run is the
  // assertDeviceReady gate (battery >= 50%, discharging, thermal NONE, screen on), recorded
  // in bound.json on a live run — never a doctor existence probe standing in for it.
  process.stdout.write(`${planned.join('\n')}\n`);
  process.exit(0);
}

if (existsSync(outDir)) {
  process.stderr.write(
    `refusing to overwrite existing output directory ${outDir}; `
    + 'pass a fresh --out per attempt (e.g. runs/candidate/a, runs/candidate/b) so prior receipts survive\n',
  );
  process.exit(2);
}
mkdirSync(outDir, { recursive: true });

// Live path only, from here down: every device contact happens below this line.
const preflightBeforeInstall = await qualifyDevice('before-install');

if (!has('--skip-install')) adbRun('install', '-r', '-d', apk);
const devicePath = adbRun('shell', 'pm', 'path', PACKAGE_NAME).split('\n')[0]?.replace(/^package:/, '') ?? '';
// The arm is proven by what the device is actually running, not by what was handed to install.
const installedSha256 = adbRun('shell', 'sha256sum', devicePath)
  .split(/\s+/)[0] ?? '';
adbRun('shell', 'am', 'force-stop', PACKAGE_NAME);

// The install heats the device, so a before-install pass must not authorise the timed launch:
// qualify again immediately before t0, and record this second reading in bound.json.
const preflightBeforeLaunch = await qualifyDevice('before-launch');

const runnerBuild = createHash('sha256').update(readFileSync(runnerCli)).digest('hex');

// PRD-360 pump-endpoint collection (option (a)). A bounded logcat collector
// runs beside the runner from before the launch (driver.prepare() clears
// logcat on launch) until after the runner exits, so the TN_PUMP_ENDPOINT
// line stamped at the displacement-correlated respond() is captured in the
// same window as the movement report and the step screenshot. The collector
// is closed reliably in every path; a missing endpoint, a prematurely
// stopped collector, or a collector error fails downstream evaluation —
// never a pass. Uses only adb logcat, the existing device transport.
const startLogcatCollector = () => {
  if (dryRun) return undefined;
  const args = ['-s', serial, 'logcat', '-v', 'brief', 'MystralColdStart:I', 'MystralRuntime:I', '*:S'];
  const child = spawn(adb, args, { stdio: ['ignore', 'pipe', 'pipe'] });
  const chunks = [];
  let error = undefined;
  child.stdout.on('data', (chunk) => chunks.push(chunk));
  child.stderr.on('data', (chunk) => chunks.push(chunk));
  child.on('error', (cause) => { error = cause; });
  return {
    child,
    chunks,
    get error() { return error; },
    stoppedEarly: false,
    stop() {
      return new Promise((resolve) => {
        if (child.exitCode !== null || child.signalCode !== null) {
          resolve();
          return;
        }
        child.on('close', () => resolve());
        child.kill('SIGTERM');
        setTimeout(() => { try { child.kill('SIGKILL'); } catch { /* exited */ } resolve(); }, 5000);
      });
    },
    text() { return Buffer.concat(chunks).toString('utf8'); },
  };
};

const t0Ns = process.hrtime.bigint();
const collector = startLogcatCollector();
const { code, stdout, stderr, timedOut } = await new Promise((settle) => {
  const child = spawn(process.execPath, runnerArgs, { cwd: repoRoot, stdio: ['ignore', 'pipe', 'pipe'] });
  const out = [];
  const err = [];
  // A bounded wait, never an open-ended one: the wall clock is the last line of defence behind the
  // runner's own --timeout and its 180s startup deadline.
  const killer = setTimeout(() => { child.kill('SIGKILL'); }, maxWallMs);
  child.stdout.on('data', (chunk) => out.push(chunk));
  child.stderr.on('data', (chunk) => err.push(chunk));
  child.on('close', (exitCode, signal) => {
    clearTimeout(killer);
    settle({
      code: exitCode ?? -1,
      stderr: Buffer.concat(err).toString('utf8'),
      stdout: Buffer.concat(out).toString('utf8'),
      timedOut: signal === 'SIGKILL',
    });
  });
});
const t1Ns = process.hrtime.bigint();
// Close the collector AFTER the runner exits so the endpoint stamped at the
// displacement report is inside the window; then freeze the log. A forced
// stop (force-stop, crash, SIGKILL) yields no endpoint line and fails R7.
let collectorError = undefined;
let collectorStoppedEarly = false;
if (collector !== undefined) {
  collectorStoppedEarly = collector.child.exitCode !== null || collector.child.signalCode !== null;
  await collector.stop();
  collectorError = collector.error?.message ?? undefined;
}
const hostLog = collector?.text() ?? '';

writeFileSync(join(outDir, 'runner-stdout.json'), stdout);
writeFileSync(join(outDir, 'runner-stderr.txt'), stderr);
// The endpoint evidence rides with the run: raw host log (logcat window),
// the exact movement response bytes the coordinator binds the endpoint to,
// and the binding metadata. Written even on failure — a missing endpoint is
// evidence, not an absence of evidence.
writeFileSync(join(outDir, 'host.log'), hostLog);
let report;
try {
  report = JSON.parse(stdout);
  writeFileSync(join(outDir, 'report.json'), `${JSON.stringify(report, null, 2)}\n`);
} catch {
  // No report is a failure the evaluator must see, not something to paper over.
  writeFileSync(join(outDir, 'report.json'), `${JSON.stringify({ pass: false, parseError: true }, null, 2)}\n`);
}
// The movement response is the step result the runner sampled AFTER input —
// the actual measured displacement, not a manually labelled payload. Bound
// by the runner's own mailbox paths: the response file the device transport
// read for the movement sample. Captured from the device even on failure;
// a missing file fails evaluation downstream.
const movementResponseDevicePath = `/sdcard/Android/data/${PACKAGE_NAME}/files/tn-playtest-response.json`;
let movementResponseBody = undefined;
let movementResponseError = undefined;
try {
  movementResponseBody = adbRun('shell', 'cat', movementResponseDevicePath);
  writeFileSync(join(outDir, 'movement-response.json'), movementResponseBody);
} catch (error) {
  movementResponseError = error instanceof Error ? error.message : String(error);
}
writeFileSync(join(outDir, 'pump-endpoint.json'), `${JSON.stringify({
  collectorError: collectorError ?? null,
  collectorStoppedEarly,
  movementResponseDevicePath,
  movementResponseError: movementResponseError ?? null,
  responsePath: movementResponseDevicePath,
}, null, 2)}\n`);

const bound = {
  activity: ACTIVITY,
  adbPath: adb,
  arm,
  apkPath: apk,
  boundMs: Number(t1Ns - t0Ns) / 1e6,
  clockSource: 'process.hrtime.bigint',
  deviceApkPath: devicePath,
  expectedSha256,
  installedSha256,
  localSha256,
  packageName: PACKAGE_NAME,
  preflight: {
    minimumBatteryPercent: MINIMUM_BATTERY_PERCENT,
    required: { charging: 'discharging', thermal: 'NONE' },
    beforeInstall: preflightBeforeInstall,
    beforeLaunch: preflightBeforeLaunch,
  },
  recordedAt: new Date().toISOString(),
  runnerCli,
  runnerCliSha256: runnerBuild,
  runnerExitCode: timedOut ? -1 : code,
  runnerTimedOut: timedOut,
  scenarioPath: scenario,
  serial,
  t0Ns: String(t0Ns),
  t1Ns: String(t1Ns),
};
writeFileSync(join(outDir, 'bound.json'), `${JSON.stringify(bound, null, 2)}\n`);
process.stdout.write(`${JSON.stringify({ artifacts: outDir, boundMs: bound.boundMs, installedSha256, runnerExitCode: bound.runnerExitCode }, null, 2)}\n`);
process.exit(timedOut || code !== 0 ? 1 : 0);
