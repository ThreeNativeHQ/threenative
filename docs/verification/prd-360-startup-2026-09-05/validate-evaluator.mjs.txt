/** A real PNG, encoded with the same library inspectFrame decodes with. */
function png(width, height, colour) {
  const image = new PNG({ height, width });
  for (let y = 0; y < height; y += 1) {
    for (let x = 0; x < width; x += 1) {
      const offset = (y * width + x) * 4;
      const [red, green, blue] = colour(x, y);
      image.data[offset] = red;
      image.data[offset + 1] = green;
      image.data[offset + 2] = blue;
      image.data[offset + 3] = 255;
    }
  }
  return PNG.sync.write(image);
}

  const failedCodes = result.checks.filter(({ pass }) => !pass).map(({ code }) => code);
  const verdictOk = result.verdict === testCase.expect;
  const rulesOk = testCase.mustFail.every((id) => failed.includes(id))
    && failed.length === testCase.mustFail.length
    && (testCase.mustFailCode === undefined || failedCodes.includes(testCase.mustFailCode));
  if (!verdictOk || !rulesOk) failures += 1;
  process.stdout.write(
    `${verdictOk && rulesOk ? 'GREEN' : 'RED  '} ${testCase.name}\n`
    + `        verdict=${result.verdict} (expected ${testCase.expect})`
    + ` boundMs=${result.boundMs} admissible=${result.admissible}\n`
    + `        failedChecks=[${failed.join(', ')}] codes=[${failedCodes.join(', ')}] required=[${testCase.mustFail.join(', ')}]\n`,
